Step 1: Verify Docker installation
docker --version
docker compose version


Step 2: Create a project directory
mkdir -p ~/loki-lab
cd ~/loki-lab


Step 3: Download Loki configuration
Download the official Loki configuration for version 3.7.0. 
Grafana Loki documentation

wget https://raw.githubusercontent.com/grafana/loki/v3.7.0/cmd/loki/loki-local-config.yaml -O loki-config.yaml


Verify the file:
ls -l



Step 4: Create docker-compose.yml

mkdir -p loki-data
sudo chown -R 10001:10001 loki-data
sudo chmod -R 755 loki-data



nano docker-compose.yml


Paste the following complete configuration:
services:
  loki:
    image: grafana/loki:3.7.0
    container_name: loki
    user: "10001:10001"
    ports:
      - "127.0.0.1:3100:3100"
    command: -config.file=/etc/loki/loki-config.yaml
    volumes:
      - ./loki-config.yaml:/etc/loki/loki-config.yaml:ro
      - ./loki-data:/tmp/loki
    restart: unless-stopped

  

volumes:
  loki_data:
  grafana_data:


Save using Ctrl+O → Enter → Ctrl+X.
Step 5: Start the containers
docker compose up -d

stop container
docker container down

docker compose run --rm --no-deps \
  --user 0 \
  --entrypoint /bin/sh loki \
  -c "mkdir -p /tmp/loki/rules && chown -R 10001:10001 /tmp/loki"

restart container 
docker compose up -d

Verify:
docker compose ps
